// GET /api/admin/batches/all-for-merge?branchId=X&year=Y&academicYear=Z
// Returns ALL batches across ALL divisions for merge selection

exports.getAllBatchesForMerge = async (req, res) => {
  try {
    const { branchId, year, academicYear } = req.query;
    
    const batches = await Batch.find({
      branch: branchId,
      year: parseInt(year),
      academicYear,
      isActive: true,
      isDeleted: { $ne: true },
      batchType: { $ne: 'merged' }  // Don't show existing merged batches
    })
      .populate('students', 'name rollNo division')
      .sort({ division: 1, name: 1 })
      .lean();
    
    // Group by division for frontend
    const grouped = {};
    batches.forEach(batch => {
      const div = batch.division || batch.divisions?.[0] || 'Unknown';
      if (!grouped[div]) grouped[div] = [];
      grouped[div].push(batch);
    });
    
    res.json({
      success: true,
      batches,
      groupedByDivision: grouped,
      totalBatches: batches.length
    });
    
  } catch (error) {
    console.error('[BATCH-MERGE] Error:', error);
    res.status(500).json({
      success: false,
      message: 'Failed to fetch batches for merge'
    });
  }
};

Naming Convention:
Year 1 (FE) → F prefix
Year 2 (SE) → S prefix
Year 3 (TE) → T prefix
Year 4 (BE) → B prefix

Division: A, B, C

Batch Number: 1, 2, 3, ...

Format: {YearPrefix}{Division}{Number}
Examples: FA1, FA2, SB3, TC1, BA3
